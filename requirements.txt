# Python 3.10+
#
# PINNED to the exact versions this project was developed and verified against,
# so a clone reproduces the reported accuracy numbers rather than something
# close to them. To relax for a different platform or Python version, loosen
# `==` to `>=` -- the floors that actually matter are noted below.
#
# opencv-contrib-python, NOT opencv-python: cv2.aruco lives in the contrib
# modules. Installing both puts two cv2 packages on the path and whichever wins
# is undefined, so if you already have opencv-python:
#
#     pip uninstall -y opencv-python opencv-python-headless
#
# The version floor that matters is 4.7: the ArUco API was restructured there
# (ArucoDetector / CharucoDetector / tuple-arg CharucoBoard) and
# cv2.aruco.estimatePoseSingleMarkers was REMOVED in 4.9. This project uses the
# modern classes plus explicit cv2.solvePnP throughout. Anything >=4.10,<5
# should work.
opencv-contrib-python==4.14.0.94

# numpy >=1.26 works; 2.x removed the 2-D np.cross overload, which this project
# does not use.
numpy==2.5.2

# scipy >=1.11 works; used only for scipy.spatial.transform.Rotation.
scipy==1.18.1

PyYAML==6.0.3
